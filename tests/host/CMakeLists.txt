cmake_minimum_required(VERSION 3.16)
project(panel_host C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
if(NOT LVGL_DIR OR NOT BOILER_FONTS)
  message(FATAL_ERROR "Pass -DLVGL_DIR=path/to/lvgl and -DBOILER_FONTS=path/to/existing/boiler_fonts.c")
endif()
file(GLOB_RECURSE LVGL_SOURCES "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_compile_definitions(lvgl PUBLIC LV_CONF_PATH=${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h)
target_include_directories(lvgl PUBLIC ${LVGL_DIR})
add_executable(panel_test host.cpp ../../src/panel_screen.cpp ${BOILER_FONTS})
target_include_directories(panel_test PRIVATE stubs ../../src)
target_link_libraries(panel_test lvgl m)
